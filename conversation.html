#!/usr/bin/env python3
"""Cohort analysis for Escapad climbing entries."""
import json
from datetime import datetime, timedelta
from collections import defaultdict

EVENTS_FILE = "/tmp/escapad_events.jsonl"
SUBS_FILE = "/tmp/escapad_subs.jsonl"

# 1. Load valid entries (Granted + ManuallyGranted)
events = []
with open(EVENTS_FILE) as f:
    for line in f:
        e = json.loads(line)
        if e["inc"] in ("Granted", "ManuallyGranted"):
            events.append({
                "cid": e["cid"],
                "name": e["name"].strip(),
                "ts": datetime.fromisoformat(e["t"].replace("Z", "+00:00")),
                "has_token_op": e["hasOp"],
            })

# 2. Group by client and sort chronologically
by_client = defaultdict(list)
for e in events:
    by_client[e["cid"]].append(e)
for cid in by_client:
    by_client[cid].sort(key=lambda x: x["ts"])

# 3. Compute features per client
clients = []
for cid, evs in by_client.items():
    visits = [e["ts"] for e in evs]
    has_carnet = any(e["has_token_op"] for e in evs)
    first_carnet = next((e["ts"] for e in evs if e["has_token_op"]), None)
    clients.append({
        "cid": cid,
        "first": visits[0],
        "n_visits": len(visits),
        "visit_2": visits[1] if len(visits) >= 2 else None,
        "visit_3": visits[2] if len(visits) >= 3 else None,
        "has_carnet": has_carnet,
        "first_carnet": first_carnet,
    })

# 4. Load subscription cards (cross-ref by client UUID)
subs_by_client = defaultdict(list)
with open(SUBS_FILE) as f:
    for line in f:
        s = json.loads(line)
        subs_by_client[s["cid"]].append({
            "plan": s["plan"],
            "createdAt": datetime.fromisoformat(s["createdAt"].replace("Z", "+00:00")),
        })

# 5. Cohort by ISO week of first visit
def week_start(dt):
    d = dt.date()
    return d - timedelta(days=d.weekday())

cohorts = defaultdict(list)
for c in clients:
    cohorts[week_start(c["first"])].append(c)

today = datetime(2026, 5, 7).date()

# 6. For each cohort, compute % at S+1, S+2, S+4, S+8, S+12
for week in sorted(cohorts.keys()):
    members = cohorts[week]
    n = len(members)
    if n < 3: continue
    age_weeks = (today - week).days / 7
    for offset in [1, 2, 4, 8, 12]:
        if age_weeks < offset: continue
        cutoff = datetime.combine(week + timedelta(weeks=offset),
                                  datetime.min.time(),
                                  tzinfo=members[0]["first"].tzinfo)
        n_2nd = sum(1 for c in members if c["visit_2"] and c["visit_2"] <= cutoff)
        n_carnet = sum(1 for c in members if c["first_carnet"] and c["first_carnet"] <= cutoff)
        n_abo = sum(1 for c in members
                    if min((s["createdAt"] for s in subs_by_client.get(c["cid"], [])),
                           default=datetime.max) <= cutoff)
        # → output the % for heatmap

#!/usr/bin/env python3
"""Consolidation finale 7 mois avec mapping produit → catégorie."""
import json, glob, os
from collections import defaultdict

CACHE = ".../tool-results/"

def normalize_cat(cat):
    if not cat: return "Autres"
    return cat.split(" / ")[0].strip()

def top_group(cat):
    cl = cat.lower().replace("é", "e")
    if "restauration" in cl: return "Restauration"
    if "boissons alcool" in cl or "vin" in cl or "biere" in cl or "bia" in cl \
       or "spiritueux" in cl or "cocktail" in cl: return "Bar"
    if "softs" in cl or "boissons chaudes" in cl: return "Soft"
    if "padel" in cl: return "Padel"
    if "escalade" in cl: return "Escalade"
    if "b2b" in cl: return "B2B"
    return "Autres"

def parse_file(file_path):
    with open(file_path) as f:
        data = json.load(f)
    obj = json.loads(data[0]["text"])
    by_cat = defaultdict(float)
    decouverte_qty = decouverte_sales = 0
    raquettes_qty = raquettes_sales = 0
    couverts_qty = couverts_sales = 0
    drinks_qty = bar_qty = 0
    carnets_qty = carnets_sales = abos_sales = 0
    for r in obj.get("rows", []):
        cat = normalize_cat(r.get("category"))
        g = top_group(cat)
        sold = float(r.get("sold", "0"))
        sales = float(r.get("total_sales", "0"))
        by_cat[g] += sales
        nl = r.get("name", "").lower()
        if ("decouverte" in nl or "découverte" in nl) and "escalade" in g.lower():
            decouverte_qty += sold; decouverte_sales += sales
        if "location de raquette" in nl:
            raquettes_qty += sold; raquettes_sales += sales
        if g == "Restauration" and any(k in nl for k in
